# Zlecanie zadań do kolejki agent-queue JEDNYM TAPEM z aplikacji GitHub (iPhone):
# GitHub app → repo homos → Actions → "Zleć zadanie serwerowi" → Run workflow →
# wybierz zadanie z listy → Run. Akcja commituje plik do .agent-queue/pending/
# na branchu agent-queue; usługa homos-runner na Windows wykona je w ≤3 min
# i odeśle wynik commitem do .agent-queue/done/ (podgląd też z telefonu).
name: Zleć zadanie serwerowi
on:
  workflow_dispatch:
    inputs:
      task:
        description: 'Zadanie (biała lista homos-runner)'
        required: true
        type: choice
        options:
          - pipeline
          - fetch:llama
          - fetch:swaps
          - scan
          - backtest
          - backtest:selection
          - backtest:validate
          - test:math
      args:
        description: 'Argumenty (spacjami, np. base-weth-usdc-030-365d)'
        required: false
        default: ''

jobs:
  enqueue:
    runs-on: ubuntu-latest
    permissions:
      contents: write
    steps:
      - uses: actions/checkout@v4
        with:
          ref: agent-queue
      - name: Dodaj zadanie do kolejki
        env:
          TASK: ${{ inputs.task }}
          ARGS: ${{ inputs.args }}
        run: |
          set -e
          id="gh-$(date -u +%Y%m%d-%H%M%S)"
          mkdir -p .agent-queue/pending
          args_json=$(python3 -c 'import json,os; print(json.dumps([a for a in os.environ.get("ARGS","").split() if a]))')
          python3 -c "import json,os; json.dump({'id': '$id', 'task': os.environ['TASK'], 'args': json.loads('$args_json'), 'source': 'github-app'}, open('.agent-queue/pending/$id.json','w'), indent=2)"
          git config user.name "homos-enqueue"
          git config user.email "actions@users.noreply.github.com"
          git add .agent-queue/pending
          git commit -m "queue: $id ${TASK} ${ARGS}"
          git push origin agent-queue
          echo "Zakolejkowano: $id → ${TASK} ${ARGS}" >> "$GITHUB_STEP_SUMMARY"
